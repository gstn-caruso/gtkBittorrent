cmake_minimum_required(VERSION 3.25)
project(GtkBitTorrent VERSION 1.0.0 LANGUAGES CXX)
set(DEFAULT_APP_VERSION "1.0.0-dev")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
  file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" DEFAULT_APP_VERSION LIMIT_COUNT 1)
endif()
set(APP_VERSION "${DEFAULT_APP_VERSION}" CACHE STRING "Artifact version")
find_package(PkgConfig REQUIRED)
pkg_check_modules(TORRENT REQUIRED IMPORTED_TARGET libtorrent-rasterbar=2.0.12)
find_package(nlohmann_json 3.11.3 REQUIRED)
add_executable(gtkbittorrent-engine engine/main.cpp)
target_compile_features(gtkbittorrent-engine PRIVATE cxx_std_20)
target_compile_options(gtkbittorrent-engine PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(gtkbittorrent-engine PRIVATE APP_VERSION="${APP_VERSION}")
target_link_libraries(gtkbittorrent-engine PRIVATE PkgConfig::TORRENT nlohmann_json::nlohmann_json)
install(TARGETS gtkbittorrent-engine RUNTIME DESTINATION bin)
